function CharacterStatus({ people }) {
  const valid = Array.isArray(people) && people.every(person =>
    person && typeof person.id === 'string' && person.id.trim() &&
    typeof person.name === 'string' && person.name.trim() && typeof person.state === 'string'
  ) && new Set(people.map(person => person.id)).size === people.length;

  if (!valid) return <p>同行者の一覧は不明です。</p>;
  if (!people.length) return <p>同行者なし。</p>;

  return (
    <dl>
      {people.map(person => (
        <div key={person.id}>
          <dt>{person.name}</dt>
          <dd>{person.state.trim() || '状態不明'}</dd>
        </div>
      ))}
    </dl>
  );
}

コード

function DraftAction() {
  const [status, setStatus] = useState('ready');

  async function addDraft() {
    setStatus('pending');
    try {
      const result = await pinx.chat.insertText('周囲を調べる');
      setStatus(result?.ok ? 'done' : 'error');
    } catch {
      setStatus('error');
    }
  }

  return (
    <section style={{ padding: 12 }}>
      <button type="button" onClick={addDraft} disabled={status === 'pending' || status === 'done'}>
        周囲を調べる
      </button>
      <div role="status">
        {status === 'done' && '下書きに入りました。確認して送信してください。'}
        {status === 'error' && '入力欄のあるプレビューチャットで試してください。'}
      </div>
    </section>
  );
}

function DraftChoices({ choices }) {
  const usable = Array.isArray(choices)
    ? choices.filter(value => typeof value === 'string').map(value => value.trim()).filter(Boolean)
    : [];
  const options = [...new Set(usable)].slice(0, 3);
  const [state, setState] = useState('ready');
  const busy = useRef(false);

  async function choose(text) {
    if (busy.current) return;
    busy.current = true;
    setState('pending');
    try {
      const result = await pinx.chat.insertText(text);
      if (result?.ok) {
        setState('done');
        return;
      }
    } catch {
      // 入力欄が使えない場合も下の表示を使う。
    }
    busy.current = false;
    setState('error');
  }

  if (!options.length) return <p>現在の行動候補はありません。</p>;

  return (
    <section aria-label="行動候補">
      <ul>
        {options.map(option => (
          <li key={option}>
            <button type="button" onClick={() => choose(option)} disabled={state === 'pending' || state === 'done'}>
              {option}
            </button>
          </li>
        ))}
      </ul>
      <p role="status">
        {state === 'done' && '下書きに入りました。確認して送信してください。'}
        {state === 'error' && '入力欄を確認して、もう一度試してください。'}
      </p>
    </section>
  );
}

function DraftBuilder({ places }) {
  const values = Array.isArray(places)
    ? places.filter(value => typeof value === 'string').map(value => value.trim()).filter(Boolean)
    : [];
  const options = [...new Set(values)];
  const purposes = ['話を聞く', '周囲を調べる'];
  const [place, setPlace] = useState('');
  const [purpose, setPurpose] = useState('');
  const [state, setState] = useState('ready');
  const busy = useRef(false);
  const selectedPlace = options.includes(place) ? place : '';
  const draft = selectedPlace && purpose ? selectedPlace + 'へ向かい、' + purpose + '。' : '';

  async function insertDraft() {
    if (!draft || busy.current) return;
    busy.current = true;
    setState('pending');
    try {
      const result = await pinx.chat.insertText(draft);
      if (result?.ok) {
        setState('done');
        return;
      }
    } catch {
      // 入力欄が使えない場合も下の表示を使う。
    }
    busy.current = false;
    setState('error');
  }

  if (!options.length) return <p>訪問先の候補がありません。</p>;

  return (
    <section>
      <p>
        <label>行き先 <select value={selectedPlace} onChange={event => setPlace(event.target.value)}>
          <option value="">選ぶ</option>
          {options.map(name => <option key={name} value={name}>{name}</option>)}
        </select></label>
      </p>
      <p>
        <label>目的 <select value={purpose} onChange={event => setPurpose(event.target.value)}>
          <option value="">選ぶ</option>
          {purposes.map(name => <option key={name} value={name}>{name}</option>)}
        </select></label>
      </p>
      {draft && <p>{draft}</p>}
      <button type="button" onClick={insertDraft} disabled={!draft || state === 'pending' || state === 'done'}>
        入力欄へ入れる
      </button>
      <p role="status">
        {state === 'done' && '下書きに入りました。確認して送信してください。'}
        {state === 'error' && '入力欄を確認して、もう一度試してください。'}
      </p>
    </section>
  );
}

function WalletLedger({ before, entries }) {
  if (!Number.isSafeInteger(before) || before < 0 || !Array.isArray(entries)) {
    return <p>所持金は不明です。</p>;
  }

  const transactions = new Map();
  for (const entry of entries) {
    if (!entry || typeof entry.id !== 'string' || !entry.id.trim() ||
        !Number.isSafeInteger(entry.delta) || !entry.delta) {
      return <p>所持金は不明です。</p>;
    }
    if (transactions.has(entry.id) && transactions.get(entry.id) !== entry.delta) {
      return <p>同じ取引の金額が一致しません。</p>;
    }
    transactions.set(entry.id, entry.delta);
  }

  let balance = before;
  for (const delta of transactions.values()) {
    balance += delta;
    if (!Number.isSafeInteger(balance)) {
      return <p>今回の取引後の所持金を確定できません。</p>;
    }
  }
  if (balance < 0) {
    return <p>今回の取引後の所持金を確定できません。</p>;
  }

  return <p>所持金 {balance} 枚</p>;
}
